option(VAVA_LIVE_AV "Enable continuous live fMP4 with FFmpeg libraries on Server" ON)
find_package(PkgConfig QUIET)
if(VAVA_LIVE_AV AND PkgConfig_FOUND)
  pkg_check_modules(LIVE_AV QUIET IMPORTED_TARGET GLOBAL libavformat>=59 libavcodec>=59 libswresample>=4 libavutil>=57)
endif()
find_package(SQLite3 3.31 REQUIRED)
add_executable(vava-server src/server.c src/legacy_import.c src/task_store.c src/file_store.c src/archive_cipher.c src/server_web.c src/web_auth.c src/json.c src/media_store.c src/live_mux.c ${PROJECT_SOURCE_DIR}/shared/ts_mux.c ${PROJECT_SOURCE_DIR}/shared/http.c ${PROJECT_SOURCE_DIR}/shared/api.c)
target_include_directories(vava-server PRIVATE ${PROJECT_SOURCE_DIR}/shared)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(vava-server PRIVATE myvava-sdk SQLite3::SQLite3 vava_protocol)
else()
  target_link_libraries(vava-server PRIVATE myvava-sdk SQLite::SQLite3 vava_protocol)
endif()
if(VAVA_LIVE_AV AND LIVE_AV_FOUND)
  target_compile_definitions(vava-server PRIVATE VAVA_LIVE_AV=1)
  target_link_libraries(vava-server PRIVATE PkgConfig::LIVE_AV)
endif()
# Stable asset output path also used by integration fixtures.
set(management_web "${CMAKE_BINARY_DIR}/server-web")
file(MAKE_DIRECTORY "${management_web}")
foreach(asset index.html management.js config.js style.css app.js api-client.js)
  configure_file("web/${asset}" "${management_web}/${asset}" COPYONLY)
endforeach()
install(DIRECTORY "${management_web}/" DESTINATION share/vava-server/web)
install(TARGETS vava-server RUNTIME DESTINATION bin)
install(PROGRAMS deploy/run-server.sh DESTINATION libexec RENAME vava-server-run)
